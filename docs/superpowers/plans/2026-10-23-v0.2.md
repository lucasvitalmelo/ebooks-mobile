> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Num PR só, corrigir os links mortos no APK (#24) e a barra de status invisível (#25), e entregar sumário, grifos com seleção nativa, barra do leitor visível ao abrir e o app inteiro seguindo o tema ativo.

**Architecture:** O que acontece dentro do livro continua na página `reader-web/` e fala com o RN só pelo protocolo de `src/reader/protocol.ts`. Links, toques em grifo e seleção passam a ser tratados com `addEventListener` dentro de cada capítulo, porque o iframe do epub.js não tem `allow-scripts` e o Chromium não executa `onclick` nele (causa do #24). A lógica isolável — resolver link, acertar toque em grifo, achatar o sumário, ordenar grifos — vai para módulos puros testados com Jest.

**Tech Stack:** Expo SDK 58 (RN 0.88), TypeScript, epub.js 0.3.93, react-native-webview 14, expo-sqlite, Zustand, Jest (jest-expo, `node:sqlite`), mais `expo-clipboard` e `expo-crypto`.

```ts
import { isExternalUrl } from './protocol'

export type LinkTarget = { kind: 'external'; url: string } | { kind: 'internal'; path: string; hash: string }

// Origem fictícia: só serve para o URL resolver caminhos relativos ao capítulo.
const BASE = 'https://livro.invalid'

export function resolveLink(raw: string, sectionPath: string): LinkTarget | null {
  const href = raw.trim()
  if (!href) return null
  if (isExternalUrl(href)) return { kind: 'external', url: href }
  if (/^[a-z][a-z0-9+.-]*:/i.test(href)) return null
  try {
    const url = new URL(href, `${BASE}${sectionPath.startsWith('/') ? '' : '/'}${sectionPath}`)
    if (url.origin !== BASE) return null
    return { kind: 'internal', path: decodeURIComponent(url.pathname), hash: url.hash }
  } catch {
    return null
  }
}
```

```ts
export interface Rect {
  left: number
  top: number
  right: number
  bottom: number
}

// slop: o dedo não acerta a borda do retângulo com precisão de pixel.
export function hitHighlight(x: number, y: number, items: { id: string; rects: Rect[] }[], slop = 4): string | null {
  for (let i = items.length - 1; i >= 0; i--) {
    const hit = items[i].rects.some(
      (r) => x >= r.left - slop && x <= r.right + slop && y >= r.top - slop && y <= r.bottom + slop,
    )
    if (hit) return items[i].id
  }
  return null
}
```

```ts
export interface TocEntry {
  label: string
  href: string
  depth: number
}

interface NavItem {
  label: string
  href: string
  subitems?: NavItem[]
}

export function flattenToc(items: NavItem[], depth = 0): TocEntry[] {
  return items.flatMap((item) => [
    { label: item.label.trim(), href: item.href, depth },
    ...flattenToc(item.subitems ?? [], depth + 1),
  ])
}

const fileOf = (href: string) => href.split('#')[0].replace(/^\.\//, '')

export function currentEntry(entries: TocEntry[], href: string | null): number {
  if (!href) return -1
  const file = fileOf(href)
  return entries.findIndex((entry) => fileOf(entry.href) === file)
}
```

```ts
// Os números do começo de um CFI, em ordem: passo do spine, caminho dentro do
// capítulo, deslocamento em caracteres. Asserções de id ([...]) e deslocamentos
// temporais ou espaciais (~, @) não contam para a ordem. Um intervalo
// (base,início,fim) começa em base + início.
export function cfiSteps(cfi: string): number[] {
  const body = cfi.replace(/^epubcfi\(/, '').replace(/\)$/, '')
  const [base, start = ''] = body.split(',')
  return (base + start)
    .replace(/\[[^\]]*\]/g, '')
    .replace(/[~@][^/!:,]*/g, '')
    .split(/[/!:]/)
    .filter((part) => part !== '')
    .map(Number)
    .filter((step) => Number.isFinite(step))
}

// Negativo quando a vem antes no livro, positivo quando b vem antes, 0 quando
// começam no mesmo lugar.
export function compareCfi(a: string, b: string): number {
  const x = cfiSteps(a)
  const y = cfiSteps(b)
  for (let i = 0; i < Math.min(x.length, y.length); i++) {
    if (x[i] !== y[i]) return x[i] - y[i]
  }
  return x.length - y.length
}

export function byBookOrder<T extends { cfiRange: string; createdAt: number }>(items: readonly T[]): T[] {
  return [...items].sort((a, b) => compareCfi(a.cfiRange, b.cfiRange) || a.createdAt - b.createdAt)
}
```

```ts
  const m = parseObject(raw)
  if (!m) return null
  switch (m.type) {
    case 'open':
      return isString(m.base64) &&
        isNullableString(m.cfi) &&
        isNullableString(m.locations) &&
        isHighlightRefs(m.highlights)
        ? {
            type: 'open',
            base64: m.base64,
            cfi: m.cfi,
            locations: m.locations,
            highlights: m.highlights.map(({ id, cfiRange }) => ({ id, cfiRange })),
          }
        : null
    case 'next':
    case 'prev':
    case 'clearSelection':
      return { type: m.type }
    case 'goTo':
      return isFilled(m.target) ? { type: 'goTo', target: m.target } : null
    case 'setTheme':
      return isReaderTheme(m.theme) ? { type: 'setTheme', theme: pickTheme(m.theme) } : null
    case 'addHighlight':
      return isFilled(m.id) && isFilled(m.cfiRange) ? { type: 'addHighlight', id: m.id, cfiRange: m.cfiRange } : null
    case 'removeHighlight':
      return isFilled(m.id) ? { type: 'removeHighlight', id: m.id } : null
    default:
      return null
  }
}

export function parseEvent(raw: string): ReaderEvent | null {
  const m = parseObject(raw)
  if (!m) return null
  switch (m.type) {
    case 'ready':
    case 'centerTap':
      return { type: m.type }
    case 'relocated':
      return isString(m.cfi) &&
        (m.percent === null || (isNumber(m.percent) && m.percent >= 0 && m.percent <= 1)) &&
        isNullableString(m.href)
        ? { type: 'relocated', cfi: m.cfi, percent: m.percent, href: m.href }
        : null
    case 'locations':
      return isString(m.data) ? { type: 'locations', data: m.data } : null
    case 'openExternal':
      return isString(m.url) && isExternalUrl(m.url) ? { type: 'openExternal', url: m.url } : null
    case 'toc':
      return isTocEntries(m.entries)
        ? { type: 'toc', entries: m.entries.map(({ label, href, depth }) => ({ label, href, depth })) }
        : null
    case 'selection':
      return isNullableString(m.cfiRange) && isString(m.text)
        ? { type: 'selection', cfiRange: m.cfiRange, text: m.text }
        : null
    case 'highlightTap':
      return isFilled(m.id) ? { type: 'highlightTap', id: m.id } : null
    case 'error':
      return isString(m.message) ? { type: 'error', message: m.message } : null
    default:
      return null
  }
}
```

```ts
function registerGestures(doc: Document, offsetX: () => number, hitTest?: (x: number, y: number) => string | null) {
  let start: { x: number; y: number; time: number; hadSelection: boolean } | null = null
  doc.addEventListener(
    'touchstart',
    (event) => {
      const touch = event.touches[0]
      start =
        event.touches.length === 1
          ? {
              x: touch.clientX,
              y: touch.clientY,
              time: Date.now(),
              hadSelection: !(doc.getSelection()?.isCollapsed ?? true),
            }
          : null
    },
    { passive: true },
  )
  doc.addEventListener('touchend', (event) => {
    const touch = event.changedTouches[0]
    const from = start
    start = null
    if (!from || !touch) return
    // Com seleção na tela, o toque só a desfaz (o sistema já cuida disso).
    if (from.hadSelection) return
    if ((event.target as Element | null)?.closest?.('a')) return
    const gesture = classifyGesture({
      dx: touch.clientX - from.x,
      dy: touch.clientY - from.y,
      x: offsetX() + touch.clientX,
      width: window.innerWidth,
      duration: Date.now() - from.time,
    })
    if (gesture !== 'next' && gesture !== 'prev') {
      const id = hitTest?.(touch.clientX, touch.clientY)
      if (id) return send({ type: 'highlightTap', id })
    }
    act(gesture)
  })
```

```tsx
import * as Clipboard from 'expo-clipboard'
import * as Crypto from 'expo-crypto'
import { router, useLocalSearchParams } from 'expo-router'
import { StatusBar } from 'expo-status-bar'
import { useCallback, useEffect, useRef, useState } from 'react'
import { Alert, AppState, Linking, Pressable, StyleSheet, Text, ToastAndroid, View } from 'react-native'
import { SafeAreaView } from 'react-native-safe-area-context'
import { AppearanceSheet } from '@/components/AppearanceSheet'
import { NavigationSheet } from '@/components/NavigationSheet'
import { Reader, type ReaderHandle } from '@/components/Reader'
import { SelectionBar } from '@/components/SelectionBar'
import { getBook, markOpened, saveLocations, type Book } from '@/lib/db/books'
import { addHighlight, deleteHighlight, listHighlights, type Highlight } from '@/lib/db/highlights'
import { getProgress, saveProgress } from '@/lib/db/progress'
import { useDb } from '@/lib/db/provider'
import { toReaderTheme } from '@/lib/db/themes'
import { readBase64 } from '@/lib/files'
import { chromeFor } from '@/lib/tone'
import { byBookOrder } from '@/reader/cfi-order'
import type { ReaderEvent } from '@/reader/protocol'
import { currentEntry, type TocEntry } from '@/reader/toc'
import { themeStore, useThemeStore } from '@/store/theme-store'

const SAVE_DELAY_MS = 1000

function report(message: string) {
  return (error: unknown) => {
    console.error(message, error)
    ToastAndroid.show(message, ToastAndroid.LONG)
  }
}

export default function BookScreen() {
  const { id } = useLocalSearchParams<{ id: string }>()
  const db = useDb()
  const reader = useRef<ReaderHandle>(null)
  const theme = useThemeStore((s) => s.active)
  const chrome = chromeFor(theme.background)
  const [book, setBook] = useState<Book | null>(null)
  const [percent, setPercent] = useState<number | null>(null)
  // Aberta ao entrar, para quem abre o livro ver que a barra existe; some no primeiro virar de página.
  const [barVisible, setBarVisible] = useState(true)
  const [sheetOpen, setSheetOpen] = useState(false)
  const [navOpen, setNavOpen] = useState(false)
  const [toc, setToc] = useState<TocEntry[]>([])
  const [currentHref, setCurrentHref] = useState<string | null>(null)
  const [highlights, setHighlights] = useState<Highlight[]>([])
  const [selection, setSelection] = useState<{ cfiRange: string; text: string } | null>(null)
  const [error, setError] = useState<string | null>(null)
  const pending = useRef<{ cfi: string; percent: number | null } | null>(null)
  // A porcentagem só existe depois das locations; um relocated sem ela não
  // pode apagar a que já se sabia.
  const lastPercent = useRef<number | null>(null)
  // O primeiro relocated depois de abrir é a posição restaurada, não um virar de página.
  const seenFirstPosition = useRef(false)
  const timer = useRef<ReturnType<typeof setTimeout> | undefined>(undefined)

  const flush = useCallback(async () => {
    clearTimeout(timer.current)
    const position = pending.current
    if (!position) return
    pending.current = null
    try {
      await saveProgress(db, { bookId: id, ...position }, Date.now())
    } catch (e) {
      // Volta para a fila, a não ser que uma posição mais nova já tenha chegado.
      pending.current ??= position
      throw e
    }
  }, [db, id])

  const save = useCallback(() => flush().catch(report('Não foi possível salvar a posição de leitura.')), [flush])

  // O sistema pode matar o app em segundo plano sem avisar: grava na hora.
  useEffect(() => {
    const subscription = AppState.addEventListener('change', (state) => {
      if (state !== 'active') void save()
    })
    return () => {
      subscription.remove()
      void save()
    }
  }, [save])

  // Também chega de novo quando a WebView é remontada (o Android matou o processo
  // dela): a posição pendente vai para o banco antes de reabrir a partir dele.
  const open = useCallback(async () => {
    try {
      await save()
      seenFirstPosition.current = false
      setSelection(null)
      const found = await getBook(db, id)
      if (!found) return setError('Livro não encontrado.')
      setBook(found)
      const [progress, base64, rows] = await Promise.all([
        getProgress(db, id),
        readBase64(found.filePath),
        listHighlights(db, id),
      ])
      lastPercent.current = progress?.percent ?? null
      setPercent(lastPercent.current)
      setHighlights(rows)
      reader.current?.send({ type: 'setTheme', theme: toReaderTheme(themeStore.getState().active) })
      reader.current?.send({
        type: 'open',
        base64,
        cfi: progress?.cfi ?? null,
        locations: found.locations,
        highlights: rows.map(({ id: highlightId, cfiRange }) => ({ id: highlightId, cfiRange })),
      })
      await markOpened(db, id, Date.now())
    } catch (e) {
      console.error(e)
      setError(e instanceof Error ? e.message : String(e))
    }
  }, [db, id, save])

  useEffect(() => {
    reader.current?.send({ type: 'setTheme', theme: toReaderTheme(theme) })
  }, [theme])

  const removeHighlight = useCallback(
    (highlight: Highlight) => {
      Alert.alert('Remover grifo?', highlight.text.slice(0, 120), [
        { text: 'Cancelar', style: 'cancel' },
        {
          text: 'Remover',
          style: 'destructive',
          onPress: () =>
            deleteHighlight(db, highlight.id)
              .then(() => {
                reader.current?.send({ type: 'removeHighlight', id: highlight.id })
                setHighlights((current) => current.filter((h) => h.id !== highlight.id))
              })
              .catch(report('Não foi possível remover o grifo.')),
        },
      ])
    },
    [db],
  )

  const onEvent = useCallback(
    (event: ReaderEvent) => {
      switch (event.type) {
        case 'ready':
          return open()
        case 'relocated': {
          if (seenFirstPosition.current) setBarVisible(false)
          seenFirstPosition.current = true
          setCurrentHref(event.href)
          if (event.percent !== null) lastPercent.current = event.percent
          pending.current = { cfi: event.cfi, percent: lastPercent.current }
          setPercent(lastPercent.current)
          clearTimeout(timer.current)
          timer.current = setTimeout(() => void save(), SAVE_DELAY_MS)
          return
        }
        case 'locations':
          return saveLocations(db, id, event.data).catch(report('Não foi possível guardar as posições do livro.'))
        case 'centerTap':
          return setBarVisible((visible) => !visible)
        case 'openExternal':
          return Linking.openURL(event.url).catch(report('Não foi possível abrir o link.'))
        case 'toc':
          return setToc(event.entries)
        case 'selection':
          return setSelection(event.cfiRange ? { cfiRange: event.cfiRange, text: event.text } : null)
        case 'highlightTap': {
          const highlight = highlights.find((h) => h.id === event.id)
          if (highlight) removeHighlight(highlight)
          return
        }
        case 'error':
          return setError(event.message)
      }
    },
    [db, id, open, save, highlights, removeHighlight],
  )

  async function onHighlight() {
    if (!selection) return
    const highlight: Highlight = {
      id: Crypto.randomUUID(),
      bookId: id,
      cfiRange: selection.cfiRange,
      text: selection.text,
      createdAt: Date.now(),
      updatedAt: Date.now(),
    }
    try {
      await addHighlight(db, highlight)
      reader.current?.send({ type: 'addHighlight', id: highlight.id, cfiRange: highlight.cfiRange })
      setHighlights((current) => byBookOrder([...current, highlight]))
    } catch (e) {
      report('Não foi possível grifar.')(e)
    }
  }

  async function onCopy() {
    if (!selection) return
    try {
      await Clipboard.setStringAsync(selection.text)
      ToastAndroid.show('Copiado', ToastAndroid.SHORT)
    } catch (e) {
      report('Não foi possível copiar.')(e)
    } finally {
      reader.current?.send({ type: 'clearSelection' })
    }
  }

  if (error) {
    return (
      <SafeAreaView style={[styles.center, { backgroundColor: theme.background }]}>
        <Text style={[styles.errorText, { color: chrome.text }]}>Não foi possível abrir o livro: {error}</Text>
        <Pressable onPress={() => router.back()} style={[styles.button, { backgroundColor: chrome.control }]}>
          <Text style={{ color: chrome.text }}>Voltar</Text>
        </Pressable>
      </SafeAreaView>
    )
  }

  return (
    <SafeAreaView style={[styles.screen, { backgroundColor: theme.background }]}>
      <StatusBar style={chrome.statusBar} />
      {/* As barras são absolutas: posicionadas dentro desta View, ficam abaixo da barra de status. */}
      <View style={styles.screen}>
        <Reader ref={reader} background={theme.background} onEvent={onEvent} />
        {barVisible && (
          <View style={[styles.bar, { backgroundColor: chrome.surface }]}>
            <Pressable onPress={() => router.back()} style={[styles.button, { backgroundColor: chrome.control }]}>
              <Text style={{ color: chrome.text }}>Voltar</Text>
            </Pressable>
            <Text style={[styles.barTitle, { color: chrome.text }]} numberOfLines={1}>
              {book?.title}
            </Text>
            <Pressable
              accessibilityLabel="Sumário e grifos"
              onPress={() => setNavOpen(true)}
              style={[styles.button, { backgroundColor: chrome.control }]}
            >
              <Text style={{ color: chrome.text }}>Sumário</Text>
            </Pressable>
            <Pressable
              accessibilityLabel="Aparência"
              onPress={() => setSheetOpen(true)}
              style={[styles.button, { backgroundColor: chrome.control }]}
            >
              <Text style={{ color: chrome.text }}>Aa</Text>
            </Pressable>
            <Text style={{ color: chrome.text }}>{percent === null ? '' : `${Math.round(percent * 100)}%`}</Text>
          </View>
        )}
        {selection && <SelectionBar chrome={chrome} onHighlight={onHighlight} onCopy={onCopy} />}
      </View>
      <AppearanceSheet visible={sheetOpen} onClose={() => setSheetOpen(false)} />
      <NavigationSheet
        visible={navOpen}
        onClose={() => setNavOpen(false)}
        chrome={chrome}
        toc={toc}
        currentIndex={currentEntry(toc, currentHref)}
        highlights={highlights}
        onGoTo={(target) => reader.current?.send({ type: 'goTo', target })}
        onRemoveHighlight={removeHighlight}
      />
    </SafeAreaView>
  )
}
```
