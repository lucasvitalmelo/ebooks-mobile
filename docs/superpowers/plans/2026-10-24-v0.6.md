```js
// Gera os PNGs do ícone a partir de assets/icon.svg. Rodar de novo só quando o SVG mudar.
import { readFileSync, writeFileSync } from 'node:fs'
import { Resvg } from '@resvg/resvg-js'

const BACKGROUND = '#1a1a1f'
const svg = readFileSync('assets/icon.svg', 'utf8')
const book = svg.match(/<g id="livro">([\s\S]*?)<\/g>/)[1]
// Centro do livro no SVG de 160: (80, 79).
const placed = (scale) => `<g transform="translate(512 512) scale(${scale}) translate(-80 -79)">${book}</g>`

// Silhueta para os ícones temáticos do Android: só o alfa conta. A capa fica
// meio transparente e as linhas e a borda da folha que vira são recortadas.
const mask = `
  <mask id="recorte" maskUnits="userSpaceOnUse" x="0" y="0" width="160" height="160">
    <rect width="160" height="160" fill="white"/>
    ${[...book.matchAll(/<path d="([^"]+)" fill="none"[^>]*>/g)]
      .map(([, d]) => `<path d="${d}" fill="none" stroke="black" stroke-width="3" stroke-linecap="round"/>`)
      .join('')}
    <path d="M80 52 C90 36 104 26 120 26 C112 50 110 84 114 108 C102 108 90 114 80 124 Z" fill="none" stroke="black" stroke-width="3"/>
  </mask>`
const silhouette = book
  .replace(/<path d="[^"]+" fill="none"[^>]*\/>/g, '')
  .replace(/fill="#e0703a"/, 'fill="white" fill-opacity="0.5"')
  .replace(/fill="#[0-9a-f]{6}"/g, 'fill="white"')

function png(name, body) {
  const doc = `<svg xmlns="http://www.w3.org/2000/svg" width="1024" height="1024" viewBox="0 0 1024 1024">${body}</svg>`
  writeFileSync(`assets/${name}`, new Resvg(doc, { fitTo: { mode: 'width', value: 1024 } }).render().asPng())
  console.log(`assets/${name}`)
}

// Ícone legado e da loja: quadrado cheio, o launcher arredonda.
png('icon.png', `<rect width="1024" height="1024" fill="${BACKGROUND}"/>${placed(5.4)}`)
// Adaptativo: o livro cabe no círculo seguro de 66/108 do quadro, uns 313 px de
// raio (a cor de fundo vem do app.json). Em 3.9 os cantos da capa ficam a ~305 px.
png('android-icon-foreground.png', placed(3.9))
png(
  'android-icon-monochrome.png',
  `<defs>${mask}</defs><g transform="translate(512 512) scale(3.9) translate(-80 -79)"><g mask="url(#recorte)">${silhouette}</g></g>`,
)
```

```ts
import type { Db } from './db'
import { getSetting, setSetting } from './settings'

export const bookDeletionKey = (id: string) => `book:${id}`
export const highlightDeletionKey = (id: string) => `highlight:${id}`

// Exclusões só são registradas depois que este aparelho conectou ao Drive: um
// livro removido antes disso não pode, mais tarde, apagar o mesmo livro que
// outro aparelho pôs no Drive (a mesma regra do leitor web).
export async function isDriveEnabled(db: Db): Promise<boolean> {
  return (await getSetting(db, 'driveEnabled')) === 'true'
}

export async function setDriveEnabled(db: Db, enabled: boolean): Promise<void> {
  await setSetting(db, 'driveEnabled', String(enabled))
}

export async function recordDeletion(db: Db, key: string, at: number): Promise<void> {
  if (!(await isDriveEnabled(db))) return
  await db.run(
    'INSERT INTO deletions (key, deletedAt) VALUES (?, ?) ON CONFLICT(key) DO UPDATE SET deletedAt = MAX(deletedAt, excluded.deletedAt)',
    [key, at],
  )
}
```

```ts
export interface Book {
  id: string
  title: string
  author: string
  // Sempre books/<id>.epub; num livro que só está no Drive o arquivo não existe.
  filePath: string
  coverPath: string | null
  size: number
  locations: string | null
  addedAt: number
  lastOpenedAt: number | null
  fontScale: number | null
  prepared: number
  // Carimbos da sincronização, os mesmos do leitor web: a última mudança no
  // registro, a última mudança no arquivo, e de quando é o arquivo que está
  // aqui (NULL: só está no Drive).
  updatedAt: number
  fileUpdatedAt: number
  blobUpdatedAt: number | null
  pendingUpload: number
  driveFileId: string | null
  driveCoverId: string | null
}

export type NewBook = Pick<Book, 'id' | 'title' | 'author' | 'filePath' | 'coverPath' | 'size' | 'addedAt'>

export const hasLocalFile = (book: Pick<Book, 'blobUpdatedAt'>) => book.blobUpdatedAt !== null

export async function insertBook(db: Db, book: NewBook): Promise<void> {
  await db.run(
```

```ts
export async function readLocalLibrary(db: Db): Promise<LocalLibrary> {
  const books = await db.getAll<Book>('SELECT * FROM books')
  const progress = await db.getAll<Progress>('SELECT * FROM progress')
  const highlights = await db.getAll<Highlight>('SELECT * FROM highlights')
  const deletions = await db.getAll<{ key: string; deletedAt: number }>('SELECT * FROM deletions')
  return {

      driveFileId: book.driveFileId,

    })),

    progress: progress.map((row) => ({
      bookId: row.bookId,
      cfi: row.cfi,
      percentage: Math.round((row.percent ?? 0) * 100),
      lastReadAt: row.updatedAt,
    })),
    highlights: highlights.map((row) => ({
      id: row.id,
      bookId: row.bookId,
      cfiRange: row.cfiRange,
      text: row.text,
      color: row.color,
      ...(row.note !== null ? { note: row.note } : {}),
      createdAt: row.createdAt,
      updatedAt: row.updatedAt,

  }

// Quantos registros a sincronização mudou aqui, por tipo (livros que chegaram
// ou saíram são anunciados), e os arquivos que ficaram sem dono: o livro e a
// capa de um livro removido, a capa antiga de um livro com capa nova.
export interface AppliedChanges {
  booksAdded: number
  booksRemoved: number
  other: number
  filesToDelete: string[]
}

// Cada registro só é gravado se ainda for o que a junção viu; um que mudou no
// meio (o leitor salvou a posição, um grifo foi editado) fica para a próxima.
export async function applyRemoteChanges(db: Db, changes: LocalChanges): Promise<AppliedChanges> {
  const applied: AppliedChanges = { booksAdded: 0, booksRemoved: 0, other: 0, filesToDelete: [] }

  for (const { id, book, expected } of changes.books) {
    const current = await getBook(db, id)
    if (expected === null) {
      if (current) continue
      await db.run(
        `INSERT INTO books (id, title, author, filePath, coverPath, size, addedAt, prepared, fontScale,
           updatedAt, fileUpdatedAt, blobUpdatedAt, pendingUpload, driveFileId, driveCoverId)
         VALUES (?, ?, ?, ?, NULL, 0, ?, 1, ?, ?, ?, NULL, 0, ?, ?)`,
        [
          id,
          book.title,
          book.author,
          bookPath(id),
          book.addedAt,
          book.fontScale ?? null,
          book.updatedAt,
          book.fileUpdatedAt,
          book.fileId,
          book.coverId,
        ],
      )
      applied.booksAdded++
      continue
    }
    if (!current || current.updatedAt !== expected) continue
    // Nenhum dos lados editou o livro: o arquivo daqui já é o conteúdo do
    // registro vencedor. Senão, um envio pendente perdeu para a edição do outro
    // lado, e o arquivo é baixado de novo para bater com o registro.
    const sameContent = current.fileUpdatedAt === current.addedAt && book.fileUpdatedAt === book.addedAt
    let blobUpdatedAt = current.blobUpdatedAt
    if (current.blobUpdatedAt !== null) {
      if (sameContent) blobUpdatedAt = book.fileUpdatedAt
      else if (current.pendingUpload === 1) blobUpdatedAt = 0
    }
    // Um registro mais novo pode trazer capa nova: ela é baixada de novo.
    const newCover = book.coverId !== null && book.updatedAt > expected
    if (newCover && current.coverPath) applied.filesToDelete.push(current.coverPath)
    await db.run(
      `UPDATE books SET title = ?, author = ?, addedAt = ?, updatedAt = ?, fileUpdatedAt = ?, driveFileId = ?,
         driveCoverId = ?, fontScale = ?, pendingUpload = 0, blobUpdatedAt = ?, coverPath = ? WHERE id = ?`,
      [
        book.title,
        book.author,
        book.addedAt,
        book.updatedAt,
        book.fileUpdatedAt,
        book.fileId,
        book.coverId,
        book.fontScale ?? current.fontScale,
        blobUpdatedAt,
        newCover ? null : current.coverPath,
        id,
      ],
    )
    applied.other++
  }

  for (const { id, expected } of changes.removedBooks) {
    const current = await getBook(db, id)
    if (!current || current.updatedAt !== expected) continue
    // Progresso e grifos saem junto pelo ON DELETE CASCADE.
    await db.run('DELETE FROM books WHERE id = ?', [id])
    applied.filesToDelete.push(current.filePath, ...(current.coverPath ? [current.coverPath] : []))
    applied.booksRemoved++
  }

  for (const { bookId, progress, expected } of changes.progress) {
    if (!(await getBook(db, bookId))) continue
    const current = await db.getFirst<{ updatedAt: number }>('SELECT updatedAt FROM progress WHERE bookId = ?', [
      bookId,

    await db.run(
      `INSERT INTO progress (bookId, cfi, percent, updatedAt) VALUES (?, ?, ?, ?)
       ON CONFLICT(bookId) DO UPDATE SET cfi = excluded.cfi, percent = excluded.percent, updatedAt = excluded.updatedAt`,
      [bookId, progress.cfi, progress.percentage / 100, progress.lastReadAt],
    )
    applied.other++
  }

  for (const { id, highlight, expected } of changes.highlights) {
    if (!(await getBook(db, highlight.bookId))) continue
    const current = await db.getFirst<{ updatedAt: number }>('SELECT updatedAt FROM highlights WHERE id = ?', [id])
    if ((current?.updatedAt ?? null) !== expected) continue
    await db.run(
      `INSERT OR REPLACE INTO highlights (id, bookId, cfiRange, text, color, note, createdAt, updatedAt)
       VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        id,
        highlight.bookId,
        highlight.cfiRange,
        highlight.text,
        highlight.color,
        highlight.note ?? null,
        highlight.createdAt,
        highlight.updatedAt,
      ],
    )
    applied.other++
  }

  for (const { id, expected } of changes.removedHighlights) {
    const current = await db.getFirst<{ updatedAt: number }>('SELECT updatedAt FROM highlights WHERE id = ?', [id])
    if (!current || current.updatedAt !== expected) continue
    await db.run('DELETE FROM highlights WHERE id = ?', [id])
    applied.other++
  }

  for (const deletion of changes.deletions) {
    await db.run(
      'INSERT INTO deletions (key, deletedAt) VALUES (?, ?) ON CONFLICT(key) DO UPDATE SET deletedAt = MAX(deletedAt, excluded.deletedAt)',
      [deletion.key, deletion.deletedAt],
    )
  }
  return applied
}
```

```ts
const API = 'https://www.googleapis.com/drive/v3'
const UPLOAD = 'https://www.googleapis.com/upload/drive/v3'
export const FOLDER_TYPE = 'application/vnd.google-apps.folder'
// Até aqui, uma requisição só; acima, uma sessão de envio (precisa do Location).
export const SIMPLE_UPLOAD_LIMIT = 5 * 1024 * 1024

export interface DriveFileRef {
  id: string
  version: string
}

export interface NewDriveFile {
  name: string
  parentId: string
  mimeType: string
  appProperties?: Record<string, string>
}

// O que a sincronização precisa do Drive: uma interface, para os testes
// rodarem contra um Drive em memória (fake-drive.ts).
export interface DriveApi {
  findFolder(name: string, parentId: string): Promise<string | null>
  createFolder(name: string, parentId: string): Promise<string>
  isFolderAlive(id: string): Promise<boolean>
  findFile(name: string, parentId: string): Promise<DriveFileRef | null>
  findBookFile(bookId: string, parentId: string): Promise<string | null>
  getVersion(id: string): Promise<string>
  createFile(file: NewDriveFile, content: Uint8Array): Promise<DriveFileRef>
  updateFile(id: string, content: Uint8Array, mimeType: string): Promise<DriveFileRef>
  downloadText(id: string): Promise<string>
  downloadBytes(id: string): Promise<Uint8Array>

export class DriveApiError extends Error {
  readonly status: number
  // O nome que o Drive dá ao problema, por exemplo storageQuotaExceeded.
  readonly reason: string | null

  constructor(status: number, reason: string | null, message: string) {
    super(message)
    this.name = 'DriveApiError'
    this.status = status
    this.reason = reason
  }
}

export function isDriveApiError(error: unknown): error is DriveApiError {
  return error instanceof Error && error.name === 'DriveApiError'
}

export interface FetchInit {
  method?: string
  headers?: Record<string, string>
  body?: string | Uint8Array
}

export interface FetchResponse {
  ok: boolean
  status: number
  statusText: string
  headers: { get(name: string): string | null }
  json(): Promise<unknown>
  text(): Promise<string>
  arrayBuffer(): Promise<ArrayBuffer>
}

export type FetchLike = (url: string, init?: FetchInit) => Promise<FetchResponse>

export interface DriveApiDeps {
  fetch: FetchLike
  getToken(): Promise<string>
  invalidateToken(): void
  log?(event: string, detail: string): void

// A linguagem de consulta do Drive usa aspas simples e escapa com \.
function quote(text: string): string {
  return `'${text.replace(/\\/g, '\\\\').replace(/'/g, "\\'")}'`
}

function concat(parts: (string | Uint8Array)[]): Uint8Array {
  const encoder = new TextEncoder()
  const chunks = parts.map((part) => (typeof part === 'string' ? encoder.encode(part) : part))
  const result = new Uint8Array(chunks.reduce((sum, chunk) => sum + chunk.length, 0))
  let offset = 0
  for (const chunk of chunks) {
    result.set(chunk, offset)
    offset += chunk.length
  }
  return result
}

export function createDriveApi(deps: DriveApiDeps): DriveApi {
  const log = deps.log ?? (() => {})

  async function errorFrom(method: string, url: string, response: FetchResponse): Promise<DriveApiError> {
    const data = (await response.json().catch(() => null)) as {
      error?: { message?: string; errors?: { reason?: string }[] }
    } | null
    const reason = data?.error?.errors?.[0]?.reason ?? null
    const message = data?.error?.message ?? response.statusText
    log('drive api: erro', `${method} ${new URL(url).pathname} ${response.status} ${reason ?? '-'} ${message}`)
    return new DriveApiError(response.status, reason, `Google Drive ${response.status}: ${message}`)
  }

  async function driveFetch(url: string, init: FetchInit = {}, retried = false): Promise<FetchResponse> {
    const headers = { ...init.headers, Authorization: `Bearer ${await deps.getToken()}` }
    const response = await deps.fetch(url, { ...init, headers })
    if (response.status === 401 && !retried) {
      deps.invalidateToken()
      return driveFetch(url, init, true)
    }
    if (!response.ok) throw await errorFrom(init.method ?? 'GET', url, response)
    return response
  }

  async function listFiles(q: string, fields: string): Promise<{ id: string; version?: string | number }[]> {
    const params = new URLSearchParams({ q, fields: `files(${fields})`, orderBy: 'createdTime', pageSize: '10' })
    const data = (await (await driveFetch(`${API}/files?${params}`)).json()) as {
      files?: { id: string; version?: string | number }[]
    }
    return data.files ?? []
  }

  async function fileRef(response: FetchResponse): Promise<DriveFileRef> {
    const data = (await response.json()) as { id: string; version: string | number }
    return { id: data.id, version: String(data.version) }
  }

  async function upload(
    method: 'POST' | 'PATCH',
    path: string,
    metadata: object,
    content: Uint8Array,
    type: string,
  ): Promise<DriveFileRef> {
    if (content.length <= SIMPLE_UPLOAD_LIMIT) {
      const boundary = `ebooks-${Math.random().toString(36).slice(2)}`
      const body = concat([
        `--${boundary}\r\nContent-Type: application/json; charset=UTF-8\r\n\r\n${JSON.stringify(metadata)}\r\n`,
        `--${boundary}\r\nContent-Type: ${type}\r\n\r\n`,
        content,
        `\r\n--${boundary}--`,
      ])
      const response = await driveFetch(`${UPLOAD}${path}?uploadType=multipart&fields=id,version`, {
        method,
        headers: { 'Content-Type': `multipart/related; boundary=${boundary}` },
        body,
      })
      return fileRef(response)
    }
    const start = await driveFetch(`${UPLOAD}${path}?uploadType=resumable&fields=id,version`, {
      method,
      headers: {
        'Content-Type': 'application/json; charset=UTF-8',
        'X-Upload-Content-Type': type,
        'X-Upload-Content-Length': String(content.length),
      },
      body: JSON.stringify(metadata),
    })
    const session = start.headers.get('Location')
    if (!session) {
      log('drive api: envio grande sem Location', `${method} ${path} ${content.length} bytes`)
      throw new DriveApiError(0, 'noUploadSession', 'O Google Drive não devolveu o endereço do envio')
    }
    return fileRef(await driveFetch(session, { method: 'PUT', headers: { 'Content-Type': type }, body: content }))
  }

  return {
    async findFolder(name, parentId) {
      const q = `name = ${quote(name)} and mimeType = '${FOLDER_TYPE}' and ${quote(parentId)} in parents and trashed = false`
      return (await listFiles(q, 'id'))[0]?.id ?? null
    },
    async createFolder(name, parentId) {
      const response = await driveFetch(`${API}/files?fields=id`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json; charset=UTF-8' },
        body: JSON.stringify({ name, mimeType: FOLDER_TYPE, parents: [parentId] }),
      })
      return ((await response.json()) as { id: string }).id
    },
    async isFolderAlive(id) {
      try {
        const response = await driveFetch(`${API}/files/${encodeURIComponent(id)}?fields=id,trashed`)
        return !((await response.json()) as { trashed?: boolean }).trashed
      } catch (error) {
        if (isDriveApiError(error) && error.status === 404) return false
        throw error
      }
    },
    async findFile(name, parentId) {
      const q = `name = ${quote(name)} and ${quote(parentId)} in parents and trashed = false`
      const file = (await listFiles(q, 'id,version'))[0]
      return file ? { id: file.id, version: String(file.version) } : null
    },
    async findBookFile(bookId, parentId) {
      const q = `appProperties has { key='bookId' and value=${quote(bookId)} } and ${quote(parentId)} in parents and trashed = false`
      return (await listFiles(q, 'id'))[0]?.id ?? null
    },
    async getVersion(id) {
      const response = await driveFetch(`${API}/files/${encodeURIComponent(id)}?fields=version`)
      return String(((await response.json()) as { version: string | number }).version)
    },
    createFile(file, content) {
      const metadata = {
        name: file.name,
        mimeType: file.mimeType,
        parents: [file.parentId],
        appProperties: file.appProperties,
      }
      return upload('POST', '/files', metadata, content, file.mimeType)
    },
    updateFile(id, content, mimeType) {
      return upload('PATCH', `/files/${encodeURIComponent(id)}`, {}, content, mimeType)
    },
    async downloadText(id) {
      return (await driveFetch(`${API}/files/${encodeURIComponent(id)}?alt=media`)).text()
    },
    async downloadBytes(id) {
      return new Uint8Array(await (await driveFetch(`${API}/files/${encodeURIComponent(id)}?alt=media`)).arrayBuffer())
    },
  }
}
```

```ts
import { saveDownloadedFile, type Book } from '@/lib/db/books'
import type { Db } from '@/lib/db/db'
import type { DriveApi } from '@/lib/drive/api'

export class BookNotHereError extends Error {
  constructor() {
    super('O arquivo do livro não está neste aparelho')
    this.name = 'BookNotHereError'
  }
}

export interface BookFileDeps {
  db: Db
  api: Pick<DriveApi, 'downloadBytes'>
  // Troca atômica (files.replaceBytes): um download pela metade não substitui o livro.
  replaceFile(path: string, bytes: Uint8Array): Promise<void>
}

// O arquivo falta (só na nuvem) ou é de antes de uma edição feita em outro aparelho.
export function needsDownload(book: Pick<Book, 'blobUpdatedAt' | 'fileUpdatedAt'>): boolean {
  return book.blobUpdatedAt === null || book.blobUpdatedAt < book.fileUpdatedAt
}

// Baixa e guarda; lança quando não dá (sem rede, Drive desconectado, arquivo sumiu).
export async function downloadBook(deps: BookFileDeps, book: Book): Promise<void> {
  if (!book.driveFileId) throw new BookNotHereError()
  const bytes = await deps.api.downloadBytes(book.driveFileId)
  await deps.replaceFile(book.filePath, bytes)
  await saveDownloadedFile(deps.db, book.id, book.fileUpdatedAt, bytes.length)
}

// Antes de abrir: baixa se precisar. Sem conexão, uma versão antiga daqui
// ainda abre (uma edição só muda título, autor e capa).
export async function ensureBookFile(deps: BookFileDeps, book: Book): Promise<void> {
  if (!needsDownload(book)) return
  try {
    await downloadBook(deps, book)
  } catch (error) {
    if (book.blobUpdatedAt !== null) return
    if (error instanceof BookNotHereError) throw error
    console.warn('Não foi possível baixar o livro', error)
    throw new BookNotHereError()
  }
}
```

```ts
import type * as SignInModule from '@react-native-google-signin/google-signin'
import { DriveAuthError } from './auth-error'

// Só os arquivos que o app criou (o mesmo escopo do leitor web, no mesmo
// projeto do Google Cloud: o "app" do drive.file é o projeto, então este
// cliente Android enxerga a pasta MyBooks que o web criou).
export const DRIVE_SCOPE = 'https://www.googleapis.com/auth/drive.file'

export interface DriveAccount {
  email: string
  name: string | null
}

let signIn: typeof SignInModule | null | undefined

// Carregado no primeiro uso: no Expo Go o módulo nativo não existe, e importar
// direto derrubaria o app na abertura. Lá o Drive fica indisponível.
function load(): typeof SignInModule {
  if (signIn === undefined) {
    try {
      // eslint-disable-next-line @typescript-eslint/no-require-imports
      signIn = require('@react-native-google-signin/google-signin') as typeof SignInModule
      signIn.GoogleSignin.configure({ scopes: [DRIVE_SCOPE] })
    } catch (error) {
      console.warn('Google Sign-In indisponível (Expo Go?)', error)
      signIn = null
    }
  }
  if (!signIn) throw new DriveAuthError('indisponivel', 'O Google Drive só funciona no APK.')
  return signIn
}

let cached: string | null = null

function accountOf(user: { email: string; name: string | null }): DriveAccount {
  return { email: user.email, name: user.name }
}

export function currentAccount(): DriveAccount | null {
  if (!signIn) return null
  const current = signIn.GoogleSignin.getCurrentUser()
  return current ? accountOf(current.user) : null
}

// Sem login guardado: desconectado. O Google recusou o login guardado (senha
// trocada, acesso revogado na conta): reconectar. O resto (sem rede, sem Play
// Services, Expo Go): indisponível.
function toAuthError(error: unknown): DriveAuthError {
  if (error instanceof DriveAuthError) return error
  if (signIn?.isErrorWithCode(error) && error.code === signIn.statusCodes.SIGN_IN_REQUIRED) {
    return new DriveAuthError('reconectar', 'Conecte o Google Drive de novo.')
  }
  return new DriveAuthError('indisponivel', 'Sem conexão com o Google.')
}

export async function getAccessToken(): Promise<string> {
  if (cached) return cached
  try {
    const { GoogleSignin, isNoSavedCredentialFoundResponse } = load()
    if (!GoogleSignin.getCurrentUser()) {
      const silent = await GoogleSignin.signInSilently()
      if (isNoSavedCredentialFoundResponse(silent))
        throw new DriveAuthError('desconectado', 'Google Drive desconectado.')
    }
    if (!GoogleSignin.getCurrentUser()?.scopes.includes(DRIVE_SCOPE)) {
      throw new DriveAuthError('reconectar', 'Conecte o Google Drive de novo.')
    }
    cached = (await GoogleSignin.getTokens()).accessToken
    return cached
  } catch (error) {
    throw toAuthError(error)
  }
}

// Depois de um 401: o Android guarda o token, e só esquecer aqui não basta.
export function invalidateAccessToken(): void {
  if (cached && signIn) void signIn.GoogleSignin.clearCachedAccessToken(cached).catch(() => {})
  cached = null
}

// null: a pessoa fechou a tela de login.
export async function connectDrive(): Promise<DriveAccount | null> {
  try {
    const { GoogleSignin, isSuccessResponse } = load()
    await GoogleSignin.hasPlayServices({ showPlayServicesUpdateDialog: true })
    const result = await GoogleSignin.signIn()
    if (!isSuccessResponse(result)) return null
    if (!result.data.scopes.includes(DRIVE_SCOPE)) {
      const granted = await GoogleSignin.addScopes({ scopes: [DRIVE_SCOPE] })
      if (!granted) return null
    }
    cached = null
    return accountOf(result.data.user)
  } catch (error) {
    throw toAuthError(error)
  }
}

// Só sai da conta neste aparelho. Não revoga no Google: revogar tira o acesso
// do projeto inteiro e desconectaria também o leitor web e os outros aparelhos.
export async function disconnectDrive(): Promise<void> {
  invalidateAccessToken()
  await load().GoogleSignin.signOut()
}
```

```ts
import { createStore } from 'zustand/vanilla'
import type { Db } from '@/lib/db/db'
import { setDriveEnabled } from '@/lib/db/deletions'
import { DriveAuthError } from '@/lib/drive/auth-error'

export type DriveStatus = 'verificando' | 'indisponivel' | 'desconectado' | 'conectado' | 'reconectar'

export interface DriveAccount {
  email: string
  name: string | null
}

export interface DriveAuth {
  getAccessToken(): Promise<string>
  currentAccount(): DriveAccount | null
  connect(): Promise<DriveAccount | null>
  disconnect(): Promise<void>
}

export interface DriveState {
  status: DriveStatus
  account: DriveAccount | null
  check(db: Db): Promise<void>
  // false: a pessoa fechou a tela de login.
  connect(db: Db): Promise<boolean>
  disconnect(db: Db): Promise<void>
}

export function createDriveStore(auth: DriveAuth) {
  // Sobe a cada desconexão: um check que começou antes e responde depois não
  // pode trazer a conexão de volta.
  let generation = 0
  return createStore<DriveState>((set) => ({
    status: 'verificando',
    account: null,
    check: async (db) => {
      const started = generation
      try {
        await auth.getAccessToken()
      } catch (error) {
        if (started !== generation) return
        set({ status: error instanceof DriveAuthError ? error.problem : 'indisponivel', account: null })
        return
      }
      if (started !== generation) return
      // Só uma desconexão explícita desliga: exclusões feitas com o login
      // momentaneamente fora ainda precisam chegar aos outros aparelhos.
      await setDriveEnabled(db, true)
      set({ status: 'conectado', account: auth.currentAccount() })
    },
    connect: async (db) => {
      const account = await auth.connect()
```

```ts
import { createStore } from 'zustand/vanilla'
import type { Book } from '@/lib/db/books'
import type { Db } from '@/lib/db/db'
import { DriveAuthError } from '@/lib/drive/auth-error'

interface DownloadState {
  downloading: string[]
  // Sobe quando um download terminou, para a biblioteca recarregar.
  revision: number
  download(db: Db, book: Book): Promise<void>
}

function reason(error: unknown): string {
  if (error instanceof DriveAuthError) {
    return error.problem === 'indisponivel' ? 'Sem conexão com a internet.' : 'Conecte o Google Drive para baixar.'
  }
  if (error instanceof TypeError || /network/i.test(String(error))) return 'Sem conexão com a internet.'
  return error instanceof Error ? error.message : String(error)
}

// Fora da tela da biblioteca, para o download continuar com um livro aberto.
export function createDownloadStore(deps: { download(db: Db, book: Book): Promise<void>; notify(text: string): void }) {
  return createStore<DownloadState>((set, get) => ({
    downloading: [],
    revision: 0,
    download: async (db, book) => {
      if (get().downloading.includes(book.id)) return
      set((state) => ({ downloading: [...state.downloading, book.id] }))
      try {
        await deps.download(db, book)
        set((state) => ({ revision: state.revision + 1 }))
      } catch (error) {
        deps.notify(`Não foi possível baixar "${book.title}". ${reason(error)}`)
      } finally {
        set((state) => ({ downloading: state.downloading.filter((id) => id !== book.id) }))
      }
    },
  }))
}
```

### Task 16: sincronização automática

**Files:**

- Create: `src/sync/useAutoSync.ts`
- Modify: `src/app/_layout.tsx`

- [ ] **Step 1: `useAutoSync.ts`**

```ts
import { useEffect } from 'react'
import { AppState } from 'react-native'
import type { Db } from '@/lib/db/db'
import { driveStore, syncStore } from '@/store/sync'

export const AUTO_SYNC_INTERVAL = 5 * 60 * 1000
// Voltar ao app logo depois de uma sincronização não dispara outra.
const MIN_GAP = 30 * 1000

let last = 0

async function runIfDue(db: Db) {
  const now = Date.now()
  if (now - last < MIN_GAP) return
  last = now
  // Aberto sem internet, o Drive ficou indisponível: verifica de novo antes,
  // senão a sessão inteira ficaria sem sincronizar.
  const status = driveStore.getState().status
  if (status === 'indisponivel' || status === 'reconectar') await driveStore.getState().check(db)
  await syncStore.getState().run(db)
}

// Ao abrir o app (depois de ver se o Drive está conectado), ao voltar para a
// frente e a cada 5 minutos com o app aberto. Em segundo plano não roda.
export function useAutoSync(db: Db) {
  useEffect(() => {
    void driveStore
      .getState()
      .check(db)
      .then(() => runIfDue(db))
    const subscription = AppState.addEventListener('change', (state) => {
      if (state === 'active') void runIfDue(db)
    })
    const timer = setInterval(() => {
      if (AppState.currentState === 'active') void runIfDue(db)
    }, AUTO_SYNC_INTERVAL)
    return () => {
      subscription.remove()
      clearInterval(timer)
    }
  }, [db])
}
```
