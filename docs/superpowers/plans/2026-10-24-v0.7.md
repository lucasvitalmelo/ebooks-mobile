> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** trocar a camada âmbar por um filtro de temperatura que mantém o preto e acrescentar o brilho do app; mostrar um anel de progresso ao abrir um livro; levar o "adicionar" para um botão flutuante; criar a tela de Configurações; e ensinar a biblioteca vazia com setas desenhadas à mão.

**Architecture:** a temperatura vira uma camada com `mixBlendMode: 'multiply'` (o preto multiplicado continua preto) e o brilho usa `expo-brightness` na janela do app, com uma camada preta extra abaixo do mínimo do sistema. O anel de progresso segue etapas reais da abertura: o app mede o download e a leitura, e a página do leitor avisa a chegada dos pedaços do livro, a abertura do EPUB e a montagem da primeira página. Configurações é uma rota nova do Expo Router que reúne Leitura, Google Drive, Biblioteca e Sobre.

**Tech Stack:** Expo SDK 58 · React Native 0.88 (`mixBlendMode`) · `expo-brightness` · `react-native-svg` · `@expo-google-fonts/caveat` · Zustand · Jest.

```ts
// A cor multiplicada sobre a tela, como o Modo Noturno do sistema: tira azul e um
// pouco de verde. Multiplicar mantém o preto preto (a camada âmbar de antes o
// clareava) e o texto escuro continua escuro.
const MAX_GREEN_CUT = 0.22
const MAX_BLUE_CUT = 0.5

export function warmthColor(percent: number): string {
  const amount = Math.min(100, Math.max(0, percent)) / 100
  const green = Math.round(255 * (1 - MAX_GREEN_CUT * amount))
  const blue = Math.round(255 * (1 - MAX_BLUE_CUT * amount))
  return `rgb(255,${green},${blue})`
}
```

- [ ] **Step 5: `src/lib/brightness.ts`**

```ts
// Abaixo deste valor a janela já está no mínimo do aparelho e o resto do
// escurecimento vem de uma camada preta: dá para ler no escuro abaixo do
// mínimo do sistema.
export const DIM_BELOW = 10
const MIN_WINDOW = 0.02
const MAX_DIM = 0.5

export interface BrightnessLevels {
  // Brilho da janela do app, de 0 a 1 (expo-brightness).
  window: number
  // Opacidade da camada preta por cima da tela.
  dim: number
}

export function brightnessLevels(percent: number): BrightnessLevels {
  const value = Math.min(100, Math.max(0, percent))
  if (value >= DIM_BELOW) return { window: value / 100, dim: 0 }
  return { window: MIN_WINDOW, dim: Math.round(((DIM_BELOW - value) / DIM_BELOW) * MAX_DIM * 100) / 100 }
}
```

```ts
// A abertura de um livro em etapas reais, cada uma com a sua fatia do anel. O
// app mede as duas primeiras; a página do leitor avisa as outras (openProgress).
export type OpenStage = 'download' | 'read' | 'transfer' | 'parse' | 'render'
export type ReaderOpenStage = Extract<OpenStage, 'transfer' | 'parse' | 'render'>

export const OPEN_STAGES: Record<OpenStage, { from: number; to: number; label: string }> = {
  download: { from: 0, to: 0.25, label: 'Baixando do Google Drive…' },
  read: { from: 0.25, to: 0.3, label: 'Lendo o arquivo…' },
  // O livro vai para a página em pedaços de 512 KB: num livro grande é a etapa mais longa.
  transfer: { from: 0.3, to: 0.7, label: 'Preparando o livro…' },
  parse: { from: 0.7, to: 0.85, label: 'Abrindo o EPUB…' },
  render: { from: 0.85, to: 1, label: 'Montando as páginas…' },
}

export function openProgress(stage: OpenStage, fraction: number): number {
  const { from, to } = OPEN_STAGES[stage]
  return from + (to - from) * Math.min(1, Math.max(0, fraction))
}

export function stageLabel(stage: OpenStage): string {
  return OPEN_STAGES[stage].label
}
```

```tsx
import Svg, { Circle } from 'react-native-svg'

const STROKE = 4
const RADIUS = (SIZE - STROKE) / 2
const LENGTH = 2 * Math.PI * RADIUS
const AnimatedCircle = Animated.createAnimatedComponent(Circle)

interface Props {
  value: number
  color: string
  track: string

export function ProgressRing({ value, color, track }: Props) {
  const [progress] = useState(() => new Animated.Value(0))
  useEffect(() => {
    Animated.timing(progress, {

  return (
    <Svg
      width={SIZE}
      height={SIZE}
      accessibilityRole="progressbar"
      accessibilityLabel={`${Math.round(value * 100)}% carregado`}
    >
      <Circle cx={SIZE / 2} cy={SIZE / 2} r={RADIUS} stroke={track} strokeWidth={STROKE} fill="none" />
      <AnimatedCircle
        cx={SIZE / 2}
        cy={SIZE / 2}
        r={RADIUS}
```

```ts
import type { Book } from './db/books'

type SpaceBook = Pick<Book, 'size' | 'blobUpdatedAt' | 'driveFileId' | 'pendingUpload'>

export function spaceUsed(books: SpaceBook[]): { bytes: number; books: number } {
  const here = books.filter((book) => book.blobUpdatedAt !== null)
  return { bytes: here.reduce((sum, book) => sum + book.size, 0), books: here.length }
}

// A mesma regra do "Remover deste aparelho": o arquivo daqui não é a única cópia.
export function canFreeSpace(book: SpaceBook): boolean {
  return book.blobUpdatedAt !== null && book.driveFileId !== null && book.pendingUpload === 0
}

const MB = 1024 * 1024

export function formatSize(bytes: number): string {
  if (bytes >= 1024 * MB) return `${(bytes / (1024 * MB)).toFixed(1).replace('.', ',')} GB`
  const mb = bytes / MB
  return `${mb >= 10 || mb === 0 ? Math.round(mb) : mb.toFixed(1).replace('.', ',')} MB`
}
```
